cmake_minimum_required(VERSION 3.16)
project(cart_solver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(cart_solver cart_solver.cpp)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(cart_solver PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(cart_solver PRIVATE /O2)
else()
  target_compile_options(cart_solver PRIVATE -O3)
endif()
